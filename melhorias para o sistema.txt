10. CORREÇÃO DO PROBLEMA: PAGAMENTO ANTECIPADO NÃO CONCLUI SERVIÇO
PROBLEMA IDENTIFICADO:
Registrar pagamento antecipado não deve marcar serviço como "Concluído".
Status do serviço (Agendado, Confirmado, Concluído, Cancelado) é independente de statusPagamento (pago, pendente).

CORREÇÃO APLICADA:
- O modal de pagamento NUNCA altera o campo "status" do agendamento.
- Apenas o botão "Concluir serviço" ou o seletor de status manual alteram "status".
- Pagamento antecipado: apenas marca como pago e gera financeiro.
- Ícone no calendário passa a ser verde (🟢) mesmo com serviço ainda agendado, indicando "já pagou, aguardando atendimento".
- Serviço concluído sem pagamento fica vermelho (🔴) e entra no alerta rolante.

function savePaymentAndClose() {
  const appointmentId = document.querySelector("#paymentAppointmentId").value;
  const appointment = state.agendamentos.find(a => a.id === appointmentId);
  if (!appointment) return;

  const method = document.querySelector("#paymentMethod").value;
  const statusPagamento = document.querySelector("#paymentStatus").value;
  const dataPagamento = document.querySelector("#paymentDate").value;
  const observacoesPagamento = document.querySelector("#paymentNotes").value;

  if (!method) { toast("Selecione a forma de pagamento."); return; }

  const { rate, taxAmount, netValue } = calculatePaymentValues(appointment.valorFinal, method);

  // Atualiza SOMENTE campos de pagamento – NÃO mexe em appointment.status
  appointment.formaPagamento = method;
  appointment.statusPagamento = statusPagamento;
  appointment.taxaPercentual = rate;
  appointment.valorTaxa = taxAmount;
  appointment.valorLiquido = netValue;
  appointment.dataPagamento = dataPagamento;
  appointment.observacoesPagamento = observacoesPagamento;

  // Gerencia financeiro
  if (statusPagamento === "pago") {
    syncAppointmentFinance(appointment); // deve usar valorLiquido
  } else {
    state.financeiro = state.financeiro.filter(f => !(f.origem === "agendamento" && f.agendamentoId === appointment.id));
    appointment.financeiroGerado = false;
  }
  save();
  renderAll();
  document.querySelector("#paymentModal").close();
  toast(`Pagamento ${statusPagamento === "pago" ? "registrado" : "marcado como pendente"}.`);
}

function syncAppointmentFinance(appointment) {
  const existing = state.financeiro.find(f => f.origem === "agendamento" && f.agendamentoId === appointment.id);
  if (appointment.statusPagamento !== "pago") {
    if (existing) {
      state.financeiro = state.financeiro.filter(f => f !== existing);
      appointment.financeiroGerado = false;
    }
    return;
  }
  const payload = {
    tipo: "entrada",
    descricao: `Atendimento - ${appointmentServiceName(appointment)} - ${appointment.nomeCliente}`,
    categoria: "Serviço",
    valor: appointment.valorLiquido,
    data: toDateInput(appointment.dataHoraInicio),
    origem: "agendamento",
    agendamentoId: appointment.id,
  };
  if (existing) {
    Object.assign(existing, payload);
  } else {
    state.financeiro.push({ id: id(), ...payload, dataCadastro: new Date().toISOString() });
    appointment.financeiroGerado = true;
  }
}

function getStatusIcon(appointment) {
  if (appointment.status === "Cancelado") return "⚫";
  const hoje = new Date();
  const dataServico = parseDate(appointment.dataHoraInicio);
  const isAtrasado = (appointment.status === "Agendado" || appointment.status === "Confirmado") && dataServico < hoje;
  if (isAtrasado) return "🔴";
  if (appointment.statusPagamento === "pago") return "🟢";
  if (appointment.status === "Concluído" && appointment.statusPagamento === "pendente") return "🔴";
  if (appointment.status === "Agendado") return "🟡";
  if (appointment.status === "Confirmado") return "🟠";
  return "⚪";
}

function updatePaymentAlert() {
  const hoje = new Date();
  const pendentes = state.agendamentos.filter(a =>
    (a.statusPagamento === "pendente" && a.status !== "Cancelado") ||
    (a.status === "Concluído" && a.statusPagamento === "pendente") ||
    ((a.status === "Agendado" || a.status === "Confirmado") && parseDate(a.dataHoraInicio) < hoje)
  );
  const alertDiv = document.querySelector("#paymentAlertBar");
  if (pendentes.length === 0) {
    alertDiv.style.display = "none";
    return;
  }
  const messages = pendentes.map(a => {
    if (parseDate(a.dataHoraInicio) < hoje && (a.status === "Agendado" || a.status === "Confirmado"))
      return `⏰ ${a.nomeCliente} - atrasado (${new Date(a.dataHoraInicio).toLocaleDateString()})`;
    if (a.statusPagamento === "pendente")
      return `⚠️ ${a.nomeCliente} - pagamento pendente`;
    return `⚠️ ${a.nomeCliente} - pendência`;
  });
  document.querySelector("#alertMessage").innerText = messages.join(" | ");
  alertDiv.style.display = "block";
}
